cmake_minimum_required(VERSION 3.24)
project(companion_native_samples LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OVPHYSX_SDK "" CACHE PATH "Installed ovphysx SDK root")
if(NOT EXISTS "${OVPHYSX_SDK}/include/ovphysx_companion/native_build.h")
    message(FATAL_ERROR "Set OVPHYSX_SDK to an installed SDK containing the native companion headers")
endif()
function(companion_plugin name source)
    add_library(${name} SHARED ${source})
    target_include_directories(${name} PRIVATE "${OVPHYSX_SDK}/include" "${OVPHYSX_SDK}/include/physx")
    target_compile_definitions(${name} PRIVATE $<$<CONFIG:Debug>:_DEBUG>)
    # All non-inline PhysX utilities come from HostApi. No PhysX/ovphysx linkage.
    set_target_properties(${name} PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
    if(UNIX AND NOT APPLE)
        target_link_options(${name} PRIVATE "-Wl,--no-undefined")
    endif()
endfunction()
companion_plugin(companion_spring spring.cpp)
companion_plugin(companion_ball ball.cpp)
companion_plugin(companion_read_only read_only.cpp)
add_executable(companion_direct_contract direct_contract.cpp)
target_include_directories(companion_direct_contract PRIVATE
    "${OVPHYSX_SDK}/include" "${OVPHYSX_SDK}/include/physx")
target_compile_definitions(companion_direct_contract PRIVATE $<$<CONFIG:Debug>:_DEBUG>)
option(COMPANION_BUILD_NEGATIVE_FIXTURES "Build rejected plugin descriptors" OFF)
if(COMPANION_BUILD_NEGATIVE_FIXTURES)
    foreach(kind VERSION BUILD COMPILER ALIGNMENT CALLBACK FLAGS BLOCK OUTPUT_PARAM)
        companion_plugin(companion_bad_${kind} spring.cpp)
        target_compile_definitions(companion_bad_${kind} PRIVATE COMPANION_TEST_BAD_${kind}=1)
    endforeach()
    foreach(kind NO_QUERIES BAD_REQUIRED BAD_BOUNDS)
        companion_plugin(companion_ball_${kind} ball.cpp)
        target_compile_definitions(companion_ball_${kind} PRIVATE COMPANION_TEST_${kind}=1)
    endforeach()
    foreach(pair "companion_callback_contract;callback_contract.cpp" "companion_ball_contract;ball_contract.cpp")
        list(GET pair 0 name)
        list(GET pair 1 source)
        add_executable(${name} ${source})
        target_include_directories(${name} PRIVATE "${OVPHYSX_SDK}/include" "${OVPHYSX_SDK}/include/physx")
        target_compile_definitions(${name} PRIVATE $<$<CONFIG:Debug>:_DEBUG>)
    endforeach()

endif()
