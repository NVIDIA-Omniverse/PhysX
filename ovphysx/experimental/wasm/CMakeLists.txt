cmake_minimum_required(VERSION 3.21)
project(OvPhysxWasmCpuPrerequisites LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(PX_GENERATE_GPU_PROJECTS OFF CACHE BOOL "" FORCE)
set(PX_GENERATE_STATIC_LIBRARIES ON CACHE BOOL "" FORCE)
set(PX_BUILDSNIPPETS OFF CACHE BOOL "" FORCE)
set(PX_BUILDPVDRUNTIME ON CACHE BOOL "" FORCE)
set(OVPHYSX_WASM_SDK_MATH "scalar" CACHE STRING "PhysX math implementation: scalar or sse2")
set_property(CACHE OVPHYSX_WASM_SDK_MATH PROPERTY STRINGS scalar sse2)
if(OVPHYSX_WASM_SDK_MATH STREQUAL "scalar")
  set(PX_SCALAR_MATH ON CACHE BOOL "" FORCE)
elseif(OVPHYSX_WASM_SDK_MATH STREQUAL "sse2")
  set(PX_SCALAR_MATH OFF CACHE BOOL "" FORCE)
  add_compile_options(-msse2)
else()
  message(FATAL_ERROR "OVPHYSX_WASM_SDK_MATH must be scalar or sse2")
endif()
set(TARGET_BUILD_PLATFORM linux CACHE STRING "" FORCE)
set(PX_OUTPUT_ARCH x86 CACHE STRING "" FORCE)
add_compile_options(-sMEMORY64=1 -pthread -msimd128 -ffp-contract=off)
if(NOT PHYSX_SOURCE_ROOT)
  message(FATAL_ERROR "Set PHYSX_SOURCE_ROOT to the official PhysX checkout")
endif()
add_compile_definitions(PX_SUPPORT_GPU_PHYSX=1)
add_subdirectory("${PHYSX_SOURCE_ROOT}/physx" sdk)
target_sources(PhysXExtensions PRIVATE "${PHYSX_SOURCE_ROOT}/physx/source/physxextensions/src/ExtDeformableSkinning.cpp")
foreach(sdk_target PVDRuntime PhysXFoundation PhysXCommon PhysX PhysXExtensions PhysXPvdSDK PhysXTask LowLevel LowLevelAABB LowLevelDynamics SceneQuery SimulationController PhysXVehicle PhysXVehicle2 PhysXCharacterKinematic)
  if(TARGET ${sdk_target})
    target_compile_options(${sdk_target} PRIVATE -Wno-error -Wno-lifetime-safety-intra-tu-suggestions -Wno-unsupported-floating-point-opt)
    if(OVPHYSX_WASM_SDK_MATH STREQUAL "scalar")
      target_compile_definitions(${sdk_target} PRIVATE PX_SIMD_DISABLED=1)
    endif()
  endif()
endforeach()
